highlight_file(__FILE__);

$password = $_POST['password'] ?? '';

$sql = "select * from users where username='$username' AND id > 0";
echo $sql . "<br>";
$data = $conn->query($sql);
if ($data && ($user = $data->fetch_assoc())) {
    $password = md5(md5($password));
    if ($password === $user['password']) {
        echo getenv("FLAG");
    }
} else {
    echo $conn->error;
}
